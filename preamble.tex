% ---------------------------------------------------------------
% Callouts in the O'Reilly note style: an icon in the left margin of
% a lightly shaded box with a colored rule.
% ---------------------------------------------------------------
\newcommand{\calloutbox}[4]{% name, title, icon, color
  \newtcolorbox{#1}[1][]{
    enhanced, breakable, sharp corners, frame hidden,
    colback=#4!5!white,
    borderline west={2pt}{0pt}{#4},
    left=3.2em, right=1em, top=1.2ex, bottom=1.2ex,
    before skip=2ex, after skip=2ex,
    fontupper=\small,
    before upper={{\sffamily\bfseries\color{#4}#2}\par\smallskip},
    overlay unbroken and first={%
      \node[anchor=north, text=#4, font=\Large]
        at ([xshift=1.7em, yshift=-1.3ex]frame.north west) {#3};},
    ##1}}
\calloutbox{aibox}{In AI}{\faRobot}{accent}

\calloutbox{tip}{Tip}{\faLightbulb[regular]}{green!45!black}
